      // Parse N<num> (المستوى), P<num> (المرحلة), SEM<num> (الأسبوع), S<num> (الحصة)
      function extractCodes(source = '') {
        const text = String(source || '');

        const nMatch = text.match(/(?:^|[^A-Za-z0-9\u0600-\u06FF])N\s*[-_:]?\s*([1-6١-٦])(?=$|[^A-Za-z0-9\u0600-\u06FF])/i);
        const pMatch = text.match(/(?:^|[^A-Za-z0-9\u0600-\u06FF])P\s*[-_:]?\s*([1-6١-٦])(?=$|[^A-Za-z0-9\u0600-\u06FF])/i);
        const semMatch = text.match(/(?:^|[^A-Za-z0-9\u0600-\u06FF])SEM\s*[-_:]?\s*([0-9١-٦]+)(?=$|[^A-Za-z0-9\u0600-\u06FF])/i);
        const sMatch = text.match(/(?:^|[^A-Za-z0-9\u0600-\u06FF])S\s*[-_:]?\s*([1-6١-٦])(?=$|[^A-Za-z0-9\u0600-\u06FF])/i);

        const levelAr = text.match(/المستوى\s*(?:الأول|الثاني|الثالث|الرابع|الخامس|السادس|1|2|3|4|5|6|١|٢|٣|٤|٥|٦)/);
        const stageAr = text.match(/المرحلة\s*(?:الأولى|الثانية|الثالثة|الرابعة|1|2|3|4|١|٢|٣|٤)/);
        const weekAr = text.match(/الأسبوع\s*([0-9١-٦]+)/);
        const sessionAr = text.match(/الحصة\s*([0-9١-٦]+)/);

        const numMap = {
          'الأول': '1', 'الأولى': '1', '1': '1', '١': '1',
          'الثاني': '2', 'الثانية': '2', '2': '2', '٢': '2',
          'الثالث': '3', 'الثالثة': '3', '3': '3', '٣': '3',
          'الرابع': '4', 'الرابعة': '4', '4': '4', '٤': '4',
          'الخامس': '5', '5': '5', '٥': '5',
          'السادس': '6', '6': '6', '٦': '6'
        };

        const norm = (raw, fallback) => {
          if (!raw) return fallback;
          const t = String(raw).trim();
          return numMap[t] || t.replace(/[٠-٩]/g, (d) => String(d.charCodeAt(0) - 1632)) || fallback;
        };

        return {
          level: nMatch ? norm(nMatch[1], '3') : norm(levelAr?.[0]?.replace(/^المستوى\s*/i, ''), '3'),
          stage: pMatch ? norm(pMatch[1], '2') : norm(stageAr?.[0]?.replace(/^المرحلة\s*/i, ''), '2'),
          week: semMatch ? norm(semMatch[1], '2') : norm(weekAr?.[1], '2'),
          session: sMatch ? norm(sMatch[1], '1') : norm(sessionAr?.[1], '1')
        };
      }

      function extractQuestions(text = '') {
        const cleaned = stripSlideMentions(text);
        const matches = cleaned.match(/[«"]?[^.!\n؟?]*[؟?][»"]?/g) || [];
        return matches
          .map((q) => q.replace(/^[«"\s-]+|[»"\s-]+$/g, '').trim())
          .filter((q) => q.length >= 4);
      }

      function extractPages(text = '') {
        const pages = [];
        const regex = /(?:ص\.?|صفحة)\s*([0-9١-٩]+(?:\s*(?:و|,|،|-)\s*[0-9١-٩]+)?)/g;
        let m;
        while ((m = regex.exec(text)) !== null) {
          pages.push(m[1].replace(/\s+/g, ' ').trim());
        }
        return Array.from(new Set(pages));
      }

      function analyzeLessonLocally(rawText, fileName = '') {
        const codes = extractCodes(`${fileName}\n${rawText}`);
        const clean = stripSlideMentions((rawText || '').trim());
        const lines = clean
          .split(/\r?\n/)
          .map((l) => l.replace(/^[◆•*▪▶\d.)\-]+\s*/, '').trim())
          .filter((l) => l.length > 1);

        const topicMatch = clean.match(/(?:موضوع\s*الدرس|عنوان\s*الدرس|الدرس\s*[0-9١-٦]*)\s*[:：\-]\s*([^\n]+)/);
        const codeLine = lines.find((l) => /\b(?:N\d+|P\d+|SEM\d+|S\d+)\b/i.test(l));
        const strippedCodeLine = codeLine
          ? codeLine
              .replace(/\b(?:N\d+|P\d+|SEM\d+|S\d+)\b/gi, '')
              .replace(/^[\s\-–—:|]+|[\s\-–—:|]+$/g, '')
              .trim()
          : '';

        const inferredTopic =
          topicMatch?.[1]?.trim() ||
          (strippedCodeLine.length > 5 ? strippedCodeLine : '') ||
          lines.find((l) => !/^(المملكة|وزارة|خطاطة|المستوى|المرحلة|الأسبوع|الحصة|الأهداف|افتتاح|النشاط الرئيسي|أنشطة التدرب|اختتام)/i.test(l) && l.length > 5) ||
          (fileName
            ? fileName
                .replace(/\.[a-z0-9]+$/i, '')
                .replace(/\b(?:N\d+|P\d+|SEM\d+|S\d+)\b/gi, '')
                .replace(/[_-]+/g, ' ')
                .trim()
            : '') ||
          DEFAULT_MINDMAP.lessonTopic;

        const objectives = [];
        let readingObjectives = false;
        for (const line of lines) {
          if (/^(الأهداف|هدف\s*الدرس|أهداف\s*الدرس)\s*[:：]?$/i.test(line)) {
            readingObjectives = true;
            continue;
          }
          if (readingObjectives) {
            if (/^(1\b|2\b|3\b|4\b|افتتاح|النشاط الرئيسي|أنشطة التدرب|اختتام\s*الدرس)/i.test(line)) {
              break;
            }
            if (line.length > 3) objectives.push(line.replace(/[:：]+$/, '').trim());
          }
        }

        if (!objectives.length) {
          const goalLines = lines.filter((l) => /(?:التعرف|تحديد|تفسير|مقارنة|استنتاج|توظيف|تمييز|فهم|وصف|اكتشاف)/i.test(l));
          if (goalLines.length) objectives.push(...goalLines.slice(0, 3));
        }

        const s1 = [], s2 = [], s3 = [], s4 = [];
        let curStage = 0;

        for (const line of lines) {
          if (/^(?:1[\s.-]*)?افتتاح\s*الدرس/i.test(line) && line.length < 55) { curStage = 1; continue; }
          if (/^(?:2[\s.-]*)?النشاط\s*الرئيسي/i.test(line) && line.length < 55) { curStage = 2; continue; }
          if (/^(?:3[\s.-]*)?أنشطة\s*التدرب/i.test(line) && line.length < 65) { curStage = 3; continue; }
          if (/^(?:4[\s.-]*)?اختتام\s*الدرس/i.test(line) && line.length < 55) { curStage = 4; continue; }

          if (curStage === 1) s1.push(line);
          else if (curStage === 2) s2.push(line);
          else if (curStage === 3) s3.push(line);
          else if (curStage === 4) s4.push(line);
        }

        if (!s1.length && !s2.length && !s3.length && !s4.length) {
          const bodyLines = lines.filter((l) => !/^(المملكة|وزارة|خطاطة|المستوى|المرحلة|الأسبوع|الحصة|الأهداف)/i.test(l));
          bodyLines.forEach((l, idx) => {
            if (/تمهيد|ملاحظة|تذكير|منزل|افتتاح/i.test(l)) s1.push(l);
            else if (/نمذجة|نشاط\s*1|نشاط\s*2|نشاط\s*3|موجهة|تجربة|مناولة/i.test(l)) s2.push(l);
            else if (/تدرب|إغناء|أتذكر|خلاصة|نشاط\s*4|نشاط\s*5|مستقلة|تطبيق/i.test(l)) s3.push(l);
            else if (/اختتام|تقويم|تعلمنا|واجب|منزل/i.test(l)) s4.push(l);
            else {
              const b = idx % 4;
              if (b === 0) s1.push(l);
              else if (b === 1) s2.push(l);
              else if (b === 2) s3.push(l);
              else s4.push(l);
            }
          });
        }

        const pages = extractPages(clean);
        const p1 = pages[0] ? `ص ${pages[0]}` : 'الكراسة';
        const p2 = pages[1] ? `ص ${pages[1]}` : p1;
        const pLast = pages[pages.length - 1] ? `ص ${pages[pages.length - 1]}` : p2;

        const allQ = extractQuestions(clean);
        const q1Arr = extractQuestions(s1.join(' '));
        const q2Arr = extractQuestions(s2.join(' '));
        const q3Arr = extractQuestions(s3.join(' '));
        const q4Arr = extractQuestions(s4.join(' '));
        const fmt = (arr) => arr.map((q) => `«${q.replace(/[«»]/g, '')}»`).join(' — ');

        const q1 = q1Arr.length
          ? `أسئلة الأستاذ في الافتتاح والتمهيد: ${fmt(q1Arr)}.`
          : `أسئلة الأستاذ في الافتتاح والتمهيد: «ماذا تلاحظون في هذا النشاط؟» — «ما العلاقة بين ملاحظتكم وموضوع «${inferredTopic}»؟».`;

        const q2 = q2Arr.length
          ? `أسئلة الأستاذ أثناء النمذجة والممارسة الموجهة: ${fmt(q2Arr)}.`
          : `أسئلة الأستاذ أثناء النمذجة والممارسة الموجهة: «كيف نفسر ما نلاحظه في «${inferredTopic}»؟» — «ما العناصر الأساسية التي توصلنا إليها؟».`;

        const q3 = q3Arr.length
          ? `أسئلة الأستاذ في التدرب والإغناء: ${fmt(q3Arr)}.`
          : `أسئلة الأستاذ في التدرب والإغناء: «كيف نطبق ما تعلمناه في حل هذا النشاط؟» — «ما الخلاصة التي نثبتها؟».`;

        const q4 = q4Arr.length
          ? `أسئلة الأستاذ الختامية للتقويم: ${fmt(q4Arr)}.`
          : (allQ.length
              ? `أسئلة الأستاذ الختامية للتقويم: «ماذا تعلمنا اليوم؟» — ${fmt(allQ.slice(-2))}.`
              : `أسئلة الأستاذ الختامية للتقويم: «ماذا تعلمنا اليوم؟» — «ما أهم فكرة ترسخت لديكم حول «${inferredTopic}»؟».`);

        return {
          ...DEFAULT_MINDMAP,
          lessonTopic: inferredTopic,
          pills: codes,
          objectives: objectives.length
            ? objectives.slice(0, 3)
            : [
                `التعرف على مضامين الدرس حول ${inferredTopic}.`,
                'توظيف الملاحظة والاستنتاج والتطبيق في الأنشطة العلمية.'
              ],
          columns: [
            {
              stage: 1,
              title: 'افتتاح الدرس',
              duration: '10 min',
              bullets: [
                ...(s1.length ? s1 : [
                  `تصحيح الأنشطة المنزلية السابقة (${p1}).`,
                  'المرور بين الصفوف والعمل في ثنائيات ومناولات بسيطة.',
                  `تقديم هدف الدرس: ${objectives[0] || inferredTopic}.`,
                  'أنشطة تمهيدية: ملاحظة الصور أو الوثائق والإجابة على اللوحة.'
                ]),
                q1
              ]
            },
            {
              stage: 2,
              title: 'النشاط الرئيسي',
              duration: '25 min',
              bullets: [
                ...(s2.length ? s2 : [
                  `عرض النمذجة أو المناولة المتعلقة بـ: ${inferredTopic}.`,
                  `ممارسة موجهة: إنجاز النشاط التطبيقي (${p2}) وتوجيه المتعلمين.`,
                  'تصحيح الأنشطة الموجهة ومناقشة النتائج.'
                ]),
                q2
              ]
            },
            {
              stage: 3,
              title: 'أنشطة التدرب والإغناء',
              duration: '20 min',
              bullets: [
                ...(s3.length ? s3 : [
                  'أتذكر: قراءة الخلاصة العلمية وتثبيت المفاهيم الأساسية.',
                  `ممارسة مستقلة (التدرب): إنجاز النشاط التطبيقي (${pLast}) وتصحيحه.`,
                  'الإغناء: إنجاز أنشطة إغنائية وتصحيح الأنشطة المنجزة.'

        };
      }

  ]

اسم الملف: ${fileName}
نص الدرس:
${lessonText}

        for (const modelName of GEMINI_FLASH_3_MODELS) {
          try {
            const url = `https://generativelanguage.googleapis.com/v1beta/models/${encodeURIComponent(modelName)}:generateContent?key=${encodeURIComponent(apiKey)}`;
            const res = await fetch(url, {
              method: 'POST',
              headers: { 'Content-Type': 'application/json' },

      async function extractTextFromFile(file) {
        const lower = (file.name || '').toLowerCase();

        if (lower.endsWith('.txt') || lower.endsWith('.md')) {
          return await file.text();
        }

        if (lower.endsWith('.pdf')) {
          if (!window.pdfjsLib) throw new Error('تعذر تحميل قارئ ملفات PDF.');
          const buf = await file.arrayBuffer();
          const pdf = await window.pdfjsLib.getDocument({ data: buf }).promise;
          const pages = [];
          for (let i = 1; i <= pdf.numPages; i++) {
            const page = await pdf.getPage(i);
            const content = await page.getTextContent();
            pages.push(content.items.map((item) => item.str).join(' '));
          }
          return pages.join('\n\n');
        }

        if (lower.endsWith('.pptx')) {
          if (!window.JSZip) throw new Error('تعذر تحميل قارئ ملفات PPTX.');
          const buf = await file.arrayBuffer();
          const zip = await window.JSZip.loadAsync(buf);
          const slidePaths = Object.keys(zip.files)
            .filter((k) => /^ppt\/slides\/slide\d+\.xml$/i.test(k))
            .sort((a, b) => {
              const na = parseInt((a.match(/(\d+)/) || [0])[1], 10);
              const nb = parseInt((b.match(/(\d+)/) || [0])[1], 10);
              return na - nb;
            });

          const collected = [];
          for (const path of slidePaths) {
            const xmlStr = await zip.files[path].async('string');
            const xmlDoc = new DOMParser().parseFromString(xmlStr, 'text/xml');
            const texts = Array.from(xmlDoc.getElementsByTagNameNS('*', 't'))
              .map((n) => n.textContent || '')
              .join(' ')
              .trim();
            if (texts) collected.push(texts);
          }
          return collected.join('\n');
        }

        if (lower.endsWith('.docx')) {
          if (window.mammoth && typeof window.mammoth.extractRawText === 'function') {
            const buf = await file.arrayBuffer();
            const res = await window.mammoth.extractRawText({ arrayBuffer: buf });
            if (res && res.value) return res.value;
          }
          if (window.JSZip) {
            const buf = await file.arrayBuffer();
            const zip = await window.JSZip.loadAsync(buf);
            if (zip.files['word/document.xml']) {
              const xmlStr = await zip.files['word/document.xml'].async('string');
              const xmlDoc = new DOMParser().parseFromString(xmlStr, 'text/xml');
              return Array.from(xmlDoc.getElementsByTagNameNS('*', 't'))
                .map((n) => n.textContent || '')
                .join(' ');
            }
          }
          throw new Error('تعذر استخراج النص من ملف DOCX.');
        }

        return await file.text();
      }

          if (!combinedSource.trim()) {
            setUploadStatus('يرجى رفع ملف (.pptx, .pdf, .txt, .md) أولاً ثم الضغط على "إنشاء خريطة ذهنية".');
            showToast('ارفع ملف الدرس أولاً', '⚠️');
            return;
          }

          setCreatingState(true);
          setUploadStatus('جاري تحليل الملف وإنشاء الخريطة الذهنية...');

          const codes = extractCodes(combinedSource);
          let mindmapData = analyzeLessonLocally(extractedLessonText || uploadedFileName, uploadedFileName);

          if (receivedApiKey) {
            const aiData = await callGeminiFlash3(receivedApiKey, combinedSource, uploadedFileName);
            if (aiData && Array.isArray(aiData.columns) && aiData.columns.length === 4) {
              mindmapData = {
                ...mindmapData,
                ...aiData,
                pills: {
                  level: codes.level || aiData?.pills?.level || mindmapData.pills.level,
                  stage: codes.stage || aiData?.pills?.stage || mindmapData.pills.stage,
                  week: codes.week || aiData?.pills?.week || mindmapData.pills.week,
                  session: codes.session || aiData?.pills?.session || mindmapData.pills.session
                }
              };
            }
          }

          renderMindMapView(mindmapData);
          showView('mindmap');
          setUploadStatus('تم إنشاء الخريطة الذهنية بنجاح.');
          showToast('تم إنشاء الخريطة الذهنية', '🧠');
        } catch (err) {
          console.error('handleCreateMindMap error:', err);
          const fallback = analyzeLessonLocally(extractedLessonText || uploadedFileName, uploadedFileName);
          renderMindMapView(fallback);
          showView('mindmap');
          showToast('تم إنشاء الخريطة الذهنية', '🧠');
        } finally {
          setCreatingState(false);
        }
      }

      async function handleFilePicked(file) {
        try {
          if (!file) return;
          uploadedFileName = file.name || '';
          extractedLessonText = '';

          const badge = document.getElementById('selectedFileBadge');
          if (badge) {
            badge.textContent = `الملف المحدد: ${uploadedFileName}`;
            badge.classList.remove('hidden');
          }
          setUploadStatus(`جاري استخراج محتوى الملف: ${uploadedFileName}...`);

          const extracted = await extractTextFromFile(file);
          extractedLessonText = (extracted || '').trim();

          const codes = extractCodes(`${uploadedFileName}\n${extractedLessonText}`);
          setUploadStatus(
            `تم تحميل "${uploadedFileName}" • المستوى ${codes.level} • المرحلة ${codes.stage} • الأسبوع ${codes.week} • الحصة ${codes.session}`
          );
          showToast(`تم تحميل: ${uploadedFileName}`, '📄');
        } catch (err) {
          console.error('handleFilePicked error:', err);
          setUploadStatus(err.message || 'تعذر قراءة الملف.');
          showToast('تعذر قراءة الملف', '⚠️');
        }
      }

      function handleResetUpload() {
        try {
          uploadedFileName = '';
          extractedLessonText = '';
          const fileInput = document.getElementById('fileInput');
          const badge = document.getElementById('selectedFileBadge');
          if (fileInput) fileInput.value = '';
          if (badge) {
            badge.textContent = '';
            badge.classList.add('hidden');
          }
          try {
            sessionStorage.removeItem('sciences_last_lesson_text');
            sessionStorage.removeItem('sciences_last_file_name');
            localStorage.removeItem('sciences_last_lesson_text');
            localStorage.removeItem('sciences_last_file_name');
          } catch (_) {}
          setUploadStatus('تمت إعادة التهيئة. جاهز لرفع ملف الدرس (.pptx, .pdf, .txt, .md).');
          showToast('Réinitialiser', '🔄');
        } catch (_) {}
      }

      function handleBackNavigation() {
        try {
          if (window.history.length > 1) {
            window.history.back();
          } else if (window.opener && !window.opener.closed) {
            window.focus();
          } else {
            showView('upload');
          }
        } catch (_) {
          showView('upload');
        }
      }

      function toggleStageHighlight(stageNum) {
        try {
          const pills = document.querySelectorAll('#stageBar [data-stage]');
          const cols = document.querySelectorAll('.stage-column[data-stage-col]');

          if (highlightedStage === stageNum) {
            highlightedStage = null;
            pills.forEach((p) => p.classList.remove('active-stage'));
            cols.forEach((c) => c.classList.remove('stage-highlight'));
            return;
          }

          highlightedStage = stageNum;
          pills.forEach((p) => p.classList.toggle('active-stage', p.getAttribute('data-stage') === stageNum));
          cols.forEach((c) => c.classList.toggle('stage-highlight', c.getAttribute('data-stage-col') === stageNum));
        } catch (_) {}
      }

      function downloadMindMapHtml() {
        try {
          const clone = document.documentElement.cloneNode(true);
          clone.querySelector('#uploadScienceView')?.classList.add('hidden');
          clone.querySelector('#resultView')?.classList.remove('hidden');
          clone.querySelector('#toastBox')?.classList.add('hidden');

          const htmlStr = '<!DOCTYPE html>\n' + clone.outerHTML;
          const blob = new Blob([htmlStr], { type: 'text/html;charset=utf-8' });
          const url = URL.createObjectURL(blob);

          const a = document.createElement('a');
          a.href = url;
          a.download = 'science-mindmap.html';
          document.body.appendChild(a);
          a.click();
          document.body.removeChild(a);
          URL.revokeObjectURL(url);

          showToast('تم تنزيل الخطاطة الذهنية', '⬇️');
        } catch (err) {
          console.error('downloadMindMapHtml error:', err);
          showToast('تعذر التنزيل', '⚠️');
        }
      }

      function bindEvents() {
        const dropZone = document.getElementById('dropZone');
        const fileInput = document.getElementById('fileInput');

        dropZone?.addEventListener('click', () => fileInput?.click());
        dropZone?.addEventListener('keydown', (e) => {
          if (e.key === 'Enter' || e.key === ' ') {
            e.preventDefault();
            fileInput?.click();
          }
        });

        dropZone?.addEventListener('dragover', (e) => {
          e.preventDefault();
          dropZone.classList.add('drag-active');
        });

        dropZone?.addEventListener('dragleave', () => {
          dropZone.classList.remove('drag-active');
        });

        dropZone?.addEventListener('drop', (e) => {
          e.preventDefault();
          dropZone.classList.remove('drag-active');
          const f = e.dataTransfer?.files?.[0];
          if (f) handleFilePicked(f);
        });

        fileInput?.addEventListener('change', (e) => {
          const f = e.target?.files?.[0];
          if (f) handleFilePicked(f);
        });

        document.getElementById('backBtn')?.addEventListener('click', handleBackNavigation);
        document.getElementById('resetUploadBtn')?.addEventListener('click', handleResetUpload);
        document.getElementById('createMindMapBtn')?.addEventListener('click', handleCreateMindMap);

        document.getElementById('backToUploadBtn')?.addEventListener('click', () => {
          showView('upload');
        });

        document.getElementById('printMindMapBtn')?.addEventListener('click', () => {
          window.print();
        });

        document.getElementById('downloadMindMapBtn')?.addEventListener('click', downloadMindMapHtml);

        document.addEventListener('click', (e) => {
          const pill = e.target.closest?.('#stageBar [data-stage]');
          if (pill) {
            if (e.target.getAttribute('contenteditable') === 'true' && document.activeElement === e.target) {
              return;
            }
            const st = pill.getAttribute('data-stage');
            if (st) toggleStageHighlight(st);
          }
        });
      }

      function init() {
        captureKeyFromSources();
        renderMindMapView(DEFAULT_MINDMAP);
        bindEvents();
      }

      if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', init);
      } else {
        init();
      }
    })();
  </script>
