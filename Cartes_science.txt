      /**
       * استخراج مفتاح API من نص أو كائن محتمل
       */
      function extractKeyFromCandidate(candidate) {
        if (!candidate) return '';
        if (typeof candidate === 'string') {
          const trimmed = candidate.trim();
          if (!trimmed) return '';
          if (trimmed.startsWith('{') && trimmed.endsWith('}')) {
            try {
              const parsed = JSON.parse(trimmed);
              return extractKeyFromCandidate(parsed);
            } catch (_) {}
          }
          if (/^AIza[0-9A-Za-z\-_]{20,}$/.test(trimmed) || trimmed.length >= 16) {
            return trimmed;
          }
          return '';
        }
        if (typeof candidate === 'object') {
          const keys = [
            API_KEY_STORAGE_NAME,
            'gemini_api_key',
            'GEMINI_API_KEY',
            'apiKey',
            'api_key',
            'API_KEY',
            'key',
          ];
          for (const k of keys) {
            if (candidate[k]) {
              const found = extractKeyFromCandidate(candidate[k]);
              if (found) return found;
            }
          }
        }
        return '';
      }

      /**
       * حفظ المفتاح في localStorage تحت الاسم الموحد sci_mindmap_gemini_key_v2
       */
      function persistApiKey(rawKey) {
        const clean = extractKeyFromCandidate(rawKey);
        if (!clean) return '';
        try {
          localStorage.setItem(API_KEY_STORAGE_NAME, clean);
        } catch (_) {}
        try {
          sessionStorage.setItem(API_KEY_STORAGE_NAME, clean);
        } catch (_) {}
        return clean;
      }

      /**
       * قراءة ومزامنة مفتاح Gemini API تلقائياً من مصادر متعددة
       */
      function syncAndGetApiKey() {
        // 1. معاملات الرابط URLSearchParams
        try {
          const params = new URLSearchParams(window.location.search || '');
          for (const paramName of ['apiKey', 'api_key', 'key', 'geminiKey']) {
            const val = params.get(paramName);
            if (val) {
              return persistApiKey(val);
            }
          }
        } catch (_) {}

        // 2. localStorage بالمفتاح الرئيسي ثم البدائل
        try {
          const directLocal = localStorage.getItem(API_KEY_STORAGE_NAME);
          if (directLocal && directLocal.trim()) {
            return directLocal.trim();
          }
          for (const altKey of ['gemini_api_key', 'GEMINI_API_KEY', 'apiKey', 'api_key', 'API_KEY']) {
            const val = localStorage.getItem(altKey);
            if (val && val.trim()) {
              return persistApiKey(val);
            }
          }
        } catch (_) {}

        // 3. sessionStorage
        try {
          for (const k of [API_KEY_STORAGE_NAME, 'gemini_api_key', 'GEMINI_API_KEY', 'apiKey', 'api_key']) {
            const val = sessionStorage.getItem(k);
            if (val && val.trim()) {
              return persistApiKey(val);
            }
          }
        } catch (_) {}

        // 4. window.name
        try {
          if (window.name) {
            const foundInName = extractKeyFromCandidate(window.name);
            if (foundInName) {
              return persistApiKey(foundInName);
            }
          }
        } catch (_) {}

        return '';
      }

      /**
       * طلب المفتاح من النافذة الأم/المنبثقة والاستماع لرسائل postMessage
       */
      function setupApiKeyHandshake() {
        syncAndGetApiKey();

        window.addEventListener('message', (event) => {
          try {
            const data = event.data;
            if (!data) return;
            if (typeof data === 'string') {
              const found = extractKeyFromCandidate(data);
              if (found) persistApiKey(found);
              return;
            }
            if (typeof data === 'object') {
              const found = extractKeyFromCandidate(data);
              if (found) persistApiKey(found);
            }
          } catch (_) {}
        });

        try {
          const requestPayload = { type: 'REQUEST_API_KEY', target: API_KEY_STORAGE_NAME };
          if (window.opener && !window.opener.closed) {
            window.opener.postMessage(requestPayload, '*');
          }
          if (window.parent && window.parent !== window) {
            window.parent.postMessage(requestPayload, '*');
          }
        } catch (_) {}
      }

      /**
       * ضبط حجم الخط داخل حاوية محددة تلقائياً حسب كمية النص
       */
      function fitContainerFontSize(container, minPx = 9.0, maxPx = 15.5) {
        if (!container) return maxPx;
        let size = maxPx;
        container.style.fontSize = `${size}px`;
        container.style.lineHeight = '1.36';

        while (container.scrollHeight > container.clientHeight && size > minPx) {
          size = +(size - 0.35).toFixed(2);
          container.style.fontSize = `${size}px`;
          if (size <= 11.5) {
            container.style.lineHeight = '1.25';
          }
        }
        return size;
      }

      function renderLines(element, lines) {

      /**
       * استخراج المستوى، المرحلة (P)، الأسبوع، ورقم الحصة (S) دون الوحدة
       */
      function extractLessonMeta(text) {
        const src = String(text || '');
        const pMatch = src.match(/\bP\s*(\d+)/i);
        const sMatch = src.match(/\bS\s*(\d+)/i);
        const semMatch = src.match(/\bsem\s*(\d+)/i);

        const levelExplicit = src.match(/المستوى\s*[:：\-]?\s*(\d+)/i);
        const stageExplicit = src.match(/المرحلة\s*[:：\-]?\s*(\d+)/i);
        const weekExplicit = src.match(/الأسبوع\s*[:：\-]?\s*(\d+)/i);
        const sessionExplicit = src.match(/(?:رقم\s*الحصة|الحصة)\s*[:：\-]?\s*(\d+)/i);

        let level = '4';
        if (levelExplicit) {
          level = levelExplicit[1];
        } else {
          const stripped = src
            .replace(/\bP\s*\d+/gi, ' ')
            .replace(/\bS\s*\d+/gi, ' ')
            .replace(/\bsem\s*\d+/gi, ' ')
            .replace(/20\d{2}\s*\/\s*20\d{2}/g, ' ');
          const numMatch = stripped.match(/(?:^|[^\d])([1-6])(?:[^\d]|$)/);
          if (numMatch) level = numMatch[1];
        }

        return {
          level,
          stage: stageExplicit ? stageExplicit[1] : (pMatch ? pMatch[1] : '1'),
          week: weekExplicit ? weekExplicit[1] : (semMatch ? semMatch[1] : '1'),
          session: sessionExplicit ? sessionExplicit[1] : (sMatch ? sMatch[1] : '1'),
        };
      }

      /**
       * صياغة هدف التعلم بصيغة تقريرية واضحة (وليس سؤالاً)
       */
      function toDeclarativeObjective(rawTopic) {
        let statement = String(rawTopic || '')
          .replace(/^(?:تقديم\s*هدف\s*الدرس|هدف\s*الدرس|أهداف\s*التعلم|الدرس)\s*[:：\-]\s*/i, '')
          .replace(/[؟?]+/g, '')
          .trim();

        if (!statement) {
          return 'تعرّف أعضاء الجهاز التنفسي ومسار الهواء أثناء التنفس';
        }

        statement = statement
          .replace(/^كيف\s+يعمل\s+/i, 'تعرّف كيفية عمل ')
          .replace(/^كيف\s+تعمل\s+/i, 'تعرّف كيفية عمل ')
          .replace(/^كيف\s+يتنفس\s+/i, 'تعرّف كيفية تنفس ')
          .replace(/^كيف\s+تتنفس\s+/i, 'تعرّف كيفية تنفس ')
          .replace(/^كيف\s+/i, 'تفسير كيفية ')
          .replace(/^ما\s+هي\s+/i, 'تحديد ')
          .replace(/^ما\s+هو\s+/i, 'تحديد ')
          .replace(/^لماذا\s+/i, 'تفسير سبب ')
          .replace(/^أين\s+/i, 'تحديد مكان ')
          .replace(/^هل\s+/i, 'التحقق من ');

        if (!/^(?:تعرّف|تعرف|تحديد|تفسير|تسمية|تصنيف|استكشاف|فهم|إبراز|مقارنة|التحقق)/i.test(statement)) {
          statement = `تعرّف ${statement}`;
        }

        return statement;
      }

      /**
       * التحليل المحلي للنص وفق القواعد التربوية الصارمة
       */
      function analyzeLessonLocally(rawText, fileName) {
        const meta = extractLessonMeta(`${fileName || ''} ${(rawText || '').slice(0, 450)}`);
        if (!rawText || !rawText.trim()) {
          return buildFallbackMindmap(meta);
        }

        const slides = rawText
          .split(/\n\s*-{3,}\s*\n|(?=\n\s*(?:شريحة|slide)\s*\d*)/gi)
          .map((s) => s.trim())
          .filter(Boolean);

        const firstSlide = slides[0] || '';
        const titleMatch =
          firstSlide.match(/عنوان\s*الدرس\s*[:：\-]\s*(.+)/i) ||
          firstSlide.match(/الدرس\s*[:：\-]\s*(.+)/i);

        const lessonTitle = titleMatch
          ? titleMatch[1].split('\n')[0].replace(/\([PS]\d+.*?\)/gi, '').trim()
          : 'كيف يعمل جهازي التنفسي؟';

        const objective = toDeclarativeObjective(lessonTitle);

        const buckets = { opening: [], main: [], practice: [], closing: [] };
        let current = null;

        for (const slide of slides) {
          if (/اختتام\s*(الدرس|درس|الحصة)?/i.test(slide)) {
            current = 'closing';
          } else if (/التدرب\s*و\s*(الإغناء|الاغناء)|الإغناء\s*و\s*(التدرب|التدرب)/i.test(slide)) {
            current = 'practice';
          } else if (/النشاط\s*(الرئيسي|رئيسي)/i.test(slide)) {
            current = 'main';
          } else if (/افتتاح\s*(الدرس|درس|الحصة)?/i.test(slide)) {
            current = 'opening';
          }
          if (current) buckets[current].push(slide);
        }

        // افتتاح الدرس: 1) تصحيح الأنشطة المنزلية 2) المرور بين الصفوف 3) تقديم هدف الدرس 4) أنشطة تمهيدية ثم أسئلة الأستاذ
        const openingRaw = splitIntoLines(
          buckets.opening
            .join('\n')
            .split('\n')
            .filter((l) => l.trim() && !/^شريحة\s*\d*/i.test(l.trim()) && !/^افتتاح\s*(الدرس|درس|الحصة)?$/i.test(l.trim()))
            .join('\n')
        );

        const homeWork = [];
        const rowWalk = [];
        const goalItem = [];
        const warmUp = [];
        const teacherQs = [];

        for (const line of openingRaw) {
          if (/^(?:تقديم\s*هدف\s*الدرس|هدف\s*الدرس|الهدف)\s*[:：]\s*/i.test(line)) {
            const cleanGoal = toDeclarativeObjective(
              line.replace(/^(?:تقديم\s*هدف\s*الدرس|هدف\s*الدرس|الهدف)\s*[:：]\s*/i, '').trim()
            );
            goalItem.push(`تقديم هدف الدرس: ${cleanGoal}`);
          } else if (/تصحيح.*منزل|أنشطة\s*منزلية|نشاط\s*منزلي/i.test(line)) {
            homeWork.push(line);
          } else if (/مرور\s*بين\s*(الصفوف|صفوف)|بين\s*(الصفوف|صفوف)/i.test(line)) {
            rowWalk.push(line);
          } else if (/أسئلة\s*الأستاذ|سؤال\s*الأستاذ/i.test(line)) {
            teacherQs.push(line);
          } else {
            warmUp.push(line);
          }
        }

        if (!rowWalk.length) rowWalk.push('المرور بين الصفوف لمتابعة إنجازات المتعلمين.');
        if (!goalItem.length) goalItem.push(`تقديم هدف الدرس: ${objective}`);
        if (!warmUp.length) warmUp.push('الأنشطة التمهيدية: ملاحظة وضعية الانطلاق ومناقشتها.');

        const openingLines = [...homeWork, ...rowWalk, ...goalItem, ...warmUp, ...teacherQs];

        // النشاط الرئيسي
        const mainRaw = splitIntoLines(
          buckets.main
            .join('\n')
            .split('\n')
            .filter((l) => l.trim() && !/^شريحة\s*\d*/i.test(l.trim()) && !/^النشاط\s*(الرئيسي|رئيسي)?$/i.test(l.trim()))
            .join('\n')
        );

        const mainLines = mainRaw.length
          ? mainRaw
          : [
              `تقديم فيديو النمذجة: ${lessonTitle}.`,
              'ممارسة موجهة: إنجاز النشاط التطبيقي بالكراسة وتحديد المطلوب.',
              'تصحيح النشاط بشكل جماعي وتدوين الخلاصات الأساسية.',
            ];

        // التدرب و الإغناء: كل نشاط متبوع في السطر الموالي مباشرة بتصحيحه دون كلمة "ثم"
        const practiceRaw = splitIntoLines(
          buckets.practice
            .join('\n')
            .split('\n')
            .filter((l) => l.trim() && !/^شريحة\s*\d*/i.test(l.trim()) && !/^(?:التدرب\s*و\s*(?:الإغناء|الاغناء)|الإغناء\s*و\s*(?:التدرب|التدرب))$/.test(l.trim()))
            .join('\n')
        );

        const practiceLines = [];
        for (const line of practiceRaw) {
          const withoutThen = line.replace(/^ثم\s+/i, '').trim();
          const actMatch = withoutThen.match(/^(?:نشاط|تمرين)\s*(?:رقم\s*)?(\d+)\s*(?:صفحة\s*(\d+))?\s*[:：\-—]?\s*(.*)$/i);
          if (actMatch) {
            const num = actMatch[1];
            const page = actMatch[2] ? ` صفحة ${actMatch[2]}` : '';
            const desc = (actMatch[3] || '').replace(/^(?:تم\s*)?تصحيح(?:ه|\s*النشاط\s*\d+)?\s*[.،-]?\s*/i, '').trim();
            practiceLines.push(desc ? `نشاط رقم ${num}${page}: ${desc}` : `نشاط رقم ${num}${page}`);
            practiceLines.push(`تصحيح النشاط ${num} ومطالبة المتعلمين بكتابة الأجوبة على الكراسة.`);
          } else if (/^(?:تم\s*)?تصحيح/i.test(withoutThen)) {
            practiceLines.push(withoutThen.replace(/^تم\s+تصحيح/i, 'تصحيح'));
          } else {
            practiceLines.push(withoutThen);
            practiceLines.push('تصحيح النشاط ومطالبة المتعلمين بتدوين الأجوبة.');
          }
        }

        if (!practiceLines.length) {
          practiceLines.push(
            'نشاط التدرب والإغناء بالكراسة: تطبيق المكتسبات الجديدة.',
            'تصحيح النشاط ومطالبة المتعلمين بتدوين الأجوبة على الكراسة.'
          );
        }

        // اختتام الدرس
        const closingRaw = splitIntoLines(
          buckets.closing
            .join('\n')
            .split('\n')
            .filter((l) => l.trim() && !/^شريحة\s*\d*/i.test(l.trim()) && !/^اختتام\s*(الدرس|درس|الحصة)?$/i.test(l.trim()))
            .join('\n')
        );

        const closingLines = closingRaw.length
          ? closingRaw
          : [
              `ماذا تعلمنا اليوم؟: ${objective}.`,
              'الأنشطة المنزلية الجديدة: إنجاز التمرين التطبيقي في الكراسة.',
            ];

        return {
          level: meta.level,
          stage: meta.stage,
          week: meta.week,
          session: meta.session,
          lessonTitle,
          objectiveLines: [objective],
          openingLines,
          mainLines,
          practiceLines,
          closingLines,
        };
      }

      /**
       * استدعاء Gemini API عبر نموذج gemini-3-flash-preview باستخدام المفتاح المخزن
       */
      async function analyzeWithGemini(apiKey, rawText, fileName) {
        const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/${GEMINI_MODEL_NAME}:generateContent?key=${encodeURIComponent(apiKey)}`;

        const systemInstruction = `
أنت خبير تربوي في مادة النشاط العلمي.
استخرج البيانات من النص بدقة وأعد JSON فقط بالبنية التالية:
{
  "level": "رقم المستوى",
  "stage": "رقم المرحلة (الرمز P)",
  "week": "رقم الأسبوع",
  "session": "رقم الحصة (الرمز S)",
  "lessonTitle": "عنوان الدرس",
  "objectiveLines": ["هدف التعلم بصيغة تقريرية وليس سؤالاً"],
  "openingLines": [
    "تصحيح الأنشطة المنزلية: ...",
    "المرور بين الصفوف ...",
    "تقديم هدف الدرس: ...",
    "الأنشطة التمهيدية: ...",
    "أسئلة الأستاذ: ..."
  ],
  "mainLines": ["عناصر النشاط الرئيسي كل عنصر في سطر مستقل"],
  "practiceLines": ["نشاط ...", "تصحيح النشاط ... بدون كلمة ثم"],
  "closingLines": ["عناصر اختتام الدرس كل عنصر في سطر مستقل"]
}
قواعد إلزامية:
- P = المرحلة، S = رقم الحصة.
- لا تذكر "الوحدة".
- صغ هدف التعلم في الدائرة المركزية بصيغة تقريرية واضحة وليس سؤالاً.
- في افتتاح الدرس التزم بالترتيب: تصحيح الأنشطة المنزلية، المرور بين الصفوف، تقديم هدف الدرس، الأنشطة التمهيدية، ثم أسئلة الأستاذ.
- في التدرب و الإغناء ضع التصحيح في السطر الموالي للنشاط مباشرة دون كلمة "ثم".
`;

        const parts = [];
        if (rawText && rawText.trim()) {
          parts.push({ text: `اسم الملف: ${fileName || 'غير محدد'}\n\nمحتوى الدرس:\n${rawText}` });
        } else {
          parts.push({ text: `اسم الملف: ${fileName || 'غير محدد'}\nحلل الصورة المرفقة واستخرج خطاطة النشاط العلمي.` });
        }

        if (state.attachedImageDataUrl && state.attachedImageMimeType) {
          const base64Data = state.attachedImageDataUrl.split(',')[1] || '';
          if (base64Data) {
            parts.push({
              inlineData: {
                mimeType: state.attachedImageMimeType,
                data: base64Data,
              },
            });
          }
        }

        const response = await fetch(endpoint, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            systemInstruction: { parts: [{ text: systemInstruction }] },
            contents: [{ role: 'user', parts }],
            generationConfig: {
              temperature: 0.2,
              responseMimeType: 'application/json',
            },
          }),
        });

        if (!response.ok) {
          throw new Error(`HTTP ${response.status}`);
        }

        const json = await response.json();
        const candidateText = json?.candidates?.[0]?.content?.parts?.[0]?.text || '';
        const cleanedJson = candidateText.replace(/^```json\s*|\s*```$/g, '').trim();
        const parsed = JSON.parse(cleanedJson);

        const meta = extractLessonMeta(`${fileName || ''} ${(rawText || '').slice(0, 300)}`);
        const title = parsed.lessonTitle || 'كيف يعمل جهازي التنفسي؟';
        const declarativeGoal = toDeclarativeObjective(
          (Array.isArray(parsed.objectiveLines) && parsed.objectiveLines[0]) || title
        );

        const cleanPractice = (Array.isArray(parsed.practiceLines) ? parsed.practiceLines : splitIntoLines(parsed.practiceLines))
          .map((line) => String(line).replace(/^ثم\s+/i, '').trim())
          .filter(Boolean);

        return {
          level: String(parsed.level || meta.level || '4'),
          stage: String(parsed.stage || meta.stage || '1'),
          week: String(parsed.week || meta.week || '1'),
          session: String(parsed.session || meta.session || '1'),
          lessonTitle: title,
          objectiveLines: [declarativeGoal],
          openingLines: Array.isArray(parsed.openingLines) ? parsed.openingLines : splitIntoLines(parsed.openingLines),
          mainLines: Array.isArray(parsed.mainLines) ? parsed.mainLines : splitIntoLines(parsed.mainLines),
          practiceLines: cleanPractice,
          closingLines: Array.isArray(parsed.closingLines) ? parsed.closingLines : splitIntoLines(parsed.closingLines),
        };
      }

      /**
       * استخراج النص من ملف PPTX
       */
      async function readPptxFile(file) {
        if (!window.JSZip) throw new Error('تعذر تحميل مكتبة قراءة PPTX');
        const buffer = await file.arrayBuffer();
        const zip = await window.JSZip.loadAsync(buffer);

        const slidePaths = Object.keys(zip.files)
          .filter((name) => /^ppt\/slides\/slide\d+\.xml$/i.test(name))
          .sort((a, b) => parseInt(a.match(/slide(\d+)\.xml/i)[1], 10) - parseInt(b.match(/slide(\d+)\.xml/i)[1], 10));

        const parser = new DOMParser();
        const slidesText = [];

        for (let i = 0; i < slidePaths.length; i++) {
          const xmlStr = await zip.files[slidePaths[i]].async('string');
          const xmlDoc = parser.parseFromString(xmlStr, 'application/xml');
          const paragraphs = Array.from(xmlDoc.getElementsByTagNameNS('*', 'p'));
          const lines = [];

          for (const p of paragraphs) {
            const txt = Array.from(p.getElementsByTagNameNS('*', 't'))
              .map((t) => t.textContent || '')
              .join('')
              .replace(/\s+/g, ' ')
              .trim();
            if (txt) lines.push(txt);
          }

          if (lines.length) {
            slidesText.push(`شريحة ${i + 1}:\n${lines.join('\n')}`);
          }
        }

        return slidesText.join('\n\n---\n\n');
      }

      /**
       * استخراج النص من ملف PDF
       */
      async function readPdfFile(file) {
        if (!window.pdfjsLib) throw new Error('تعذر تحميل مكتبة قراءة PDF');
        const buffer = await file.arrayBuffer();
        const doc = await window.pdfjsLib.getDocument({ data: buffer }).promise;
        const pagesText = [];

        for (let p = 1; p <= doc.numPages; p++) {
          const page = await doc.getPage(p);
          const content = await page.getTextContent();
          const line = content.items
            .map((item) => item.str)
            .join(' ')
            .replace(/\s+/g, ' ')
            .trim();
          if (line) {
            pagesText.push(`شريحة ${p}:\n${line}`);
          }
        }

        return pagesText.join('\n\n---\n\n');
      }

      /**
       * قراءة صورة وتحويلها إلى Data URL
       */
      function readImageAsDataUrl(file) {
        return new Promise((resolve, reject) => {
          const reader = new FileReader();
          reader.onload = () => resolve(String(reader.result || ''));
          reader.onerror = () => reject(new Error('تعذر قراءة الصورة'));
          reader.readAsDataURL(file);
        });
      }

      /**
       * معالجة الملف المرفوع في صفحة التحميل
       */
      async function handleUploadedFile(file) {
        if (!file) return;

        const statusBar = document.getElementById('file-status-bar');
        const statusText = document.getElementById('file-status-text');
        const hiddenTextarea = document.getElementById('extracted-lesson-text');

        state.fileName = file.name;
        state.attachedImageDataUrl = '';
        state.attachedImageMimeType = '';

        statusBar.classList.remove('hidden');
        statusText.textContent = `جاري قراءة الملف: ${file.name}...`;

        try {
          const lower = file.name.toLowerCase();
          if (lower.endsWith('.pptx')) {
            hiddenTextarea.value = await readPptxFile(file);
            statusText.textContent = `الملف جاهز: ${file.name}`;
            showToast('تم استخراج نصوص ملف PPTX بنجاح', 'success');
          } else if (lower.endsWith('.pdf')) {
            hiddenTextarea.value = await readPdfFile(file);
            statusText.textContent = `الملف جاهز: ${file.name}`;
            showToast('تم استخراج نصوص ملف PDF بنجاح', 'success');
          } else if (file.type && file.type.startsWith('image/')) {
            state.attachedImageDataUrl = await readImageAsDataUrl(file);
            state.attachedImageMimeType = file.type;
            statusText.textContent = `الصورة جاهزة: ${file.name}`;
            showToast('تم تحميل الصورة بنجاح', 'success');
          } else {
            hiddenTextarea.value = await file.text();
            statusText.textContent = `الملف جاهز: ${file.name}`;
            showToast('تمت قراءة الملف بنجاح', 'success');
          }
        } catch (err) {
          statusText.textContent = `تم اختيار: ${file.name}`;
          showToast(err.message || 'سيتم استخدام التحليل المحلي للملف', 'info');
        }
      }

      /**
       * تنفيذ التحليل والانتقال إلى صفحة الخطاطة الذهنية
       */
      async function handleGenerateClick() {
        const btn = document.getElementById('btn-create-mindmap');
        const spinner = document.getElementById('loading-spinner');
        const label = document.getElementById('create-btn-text');
        const rawText = document.getElementById('extracted-lesson-text').value.trim();

        btn.disabled = true;
        spinner.classList.remove('hidden');
        label.textContent = 'جاري إنشاء الخريطة الذهنية...';

        const apiKey = syncAndGetApiKey();
        let mindmapData = null;

        if (apiKey) {
          try {
            mindmapData = await analyzeWithGemini(apiKey, rawText, state.fileName);
          } catch (_) {
            mindmapData = null;
          }
        }

        if (!mindmapData) {
          mindmapData = analyzeLessonLocally(rawText, state.fileName);
        }

        renderMindmap(mindmapData);
        showPage('mindmap');
        showToast('تم إنشاء الخطاطة الذهنية بنجاح', 'success');

        btn.disabled = false;
        spinner.classList.add('hidden');
        label.textContent = 'إنشاء خريطة ذهنية';
      }

            e.preventDefault();
            e.stopPropagation();

          });

            e.preventDefault();
